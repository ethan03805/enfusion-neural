cmake_minimum_required(VERSION 3.20)
project(EnfusionNeural LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The native reference backend currently requires Windows and D3D12")
endif()
add_executable(enr_gpu enr_gpu.cpp)
target_compile_features(enr_gpu PRIVATE cxx_std_17)
target_compile_definitions(enr_gpu PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(enr_gpu PRIVATE d3d12 dxgi d3dcompiler)

add_executable(enr_companion companion.cpp)
target_compile_features(enr_companion PRIVATE cxx_std_20)
set_property(TARGET enr_companion PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_definitions(enr_companion PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(enr_companion PRIVATE d3d11 dxgi d3dcompiler windowsapp dwmapi dcomp)

add_executable(enr_curve_smoke curve_smoke.cpp)
target_compile_features(enr_curve_smoke PRIVATE cxx_std_20)
target_compile_definitions(enr_curve_smoke PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(enr_curve_smoke PRIVATE d3d11 dxgi d3dcompiler windowsapp)

add_executable(enr_adapter_info adapter_info.cpp)
target_compile_features(enr_adapter_info PRIVATE cxx_std_17)
target_link_libraries(enr_adapter_info PRIVATE dxgi)

add_executable(enr_dce_replay dce_replay.cpp)
target_compile_features(enr_dce_replay PRIVATE cxx_std_20)
target_compile_definitions(enr_dce_replay PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(enr_dce_replay PRIVATE d3d11 dxgi d3dcompiler windowsapp)
